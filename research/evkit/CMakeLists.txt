cmake_minimum_required(VERSION 3.18)
project(evkit LANGUAGES CXX)

option(EVKIT_WITH_CUDA "Build the CUDA kernels (needs nvcc or clang CUDA)" OFF)
option(EVKIT_BUILD_TESTS "Build C++ tests" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -fno-math-errno")
find_package(Threads REQUIRED)

add_library(evkit STATIC src/events.cpp src/evb.cpp src/tokenize.cpp src/ssm_cpu.cpp)
target_include_directories(evkit PUBLIC include)
target_link_libraries(evkit PUBLIC Threads::Threads)
set_target_properties(evkit PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(EVKIT_WITH_CUDA)
  enable_language(CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 80 86 89 90)
  endif()
  add_library(evkit_cuda STATIC cuda/ssm_scan.cu cuda/readout.cu cuda/tokenize.cu)
  target_include_directories(evkit_cuda PUBLIC include)
  target_compile_options(evkit_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3 --use_fast_math -lineinfo>)
  set_target_properties(evkit_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_SEPARABLE_COMPILATION OFF)
  add_executable(test_cuda tests/test_cuda.cu)
  target_link_libraries(test_cuda evkit_cuda evkit)
  add_executable(bench_cuda bench/bench_cuda.cu)
  target_link_libraries(bench_cuda evkit_cuda evkit)
endif()

add_executable(evkit-convert tools/evkit_convert.cpp)
target_link_libraries(evkit-convert evkit)
add_executable(bench_decode bench/bench_decode.cpp)
target_link_libraries(bench_decode evkit)

if(EVKIT_BUILD_TESTS)
  enable_testing()
  foreach(t test_philox test_decoders test_evb test_tokenize test_ssm_cpu)
    add_executable(${t} tests/${t}.cpp)
    target_link_libraries(${t} evkit)
    add_test(NAME ${t} COMMAND ${t})
  endforeach()
endif()
